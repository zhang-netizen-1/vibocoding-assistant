npm install @forma/ui

点击复制，将示例命令写入剪贴板。

使用方式VANILLA JAVASCRIPT 
// 挂载到任意容器，onClick 支持同步或异步操作。
const saveButton = new Button(document.querySelector('#container'), {
  label: '保存更改',
  variant: 'primary', // primary | secondary | danger
  onClick: async () => { await saveChanges(); }
});

saveButton.setDisabled(true);
saveButton.setLoading(true);
saveButton.setLabel('确认保存');
saveButton.setVariant('secondary');

// 仅在复制成功后显示反馈，失败时提示重试。
new Button(document.querySelector('#copy'), {
  label: '复制命令',
  variant: 'secondary',
  successLabel: '已复制',
  onClick: () => copyText('npm install @forma/ui')
});
