接入与设计令牌 同一个 Button 类，可独立复用
// 连同 :root 令牌与 .ui-button 样式一起复用，无需安装依赖。
const button = new Button({
  label: '保存更改',
  variant: 'primary', // primary | secondary | danger
  loadingLabel: '保存中…',
  onClick: async () => { await saveChanges(); }
});
document.querySelector('#target').append(button.element);
button.setDisabled(true);       // 原生禁用
button.setDisabled(false);
button.setLoading(true);        // 外部控制加载
button.setLoading(false);
button.setOptions({ label: '再次保存', variant: 'secondary' });

// 复制类操作：Promise 成功后才展示成功态，失败可重试。
const copy = new Button({
  label: '复制文本', icon: 'copy', variant: 'secondary',
  successLabel: '已复制', successDuration: 1800,
  onClick: () => copyText('要复制的内容')
});
--accent 主操作--danger 危险操作--focus 焦点环--radius-md 圆角 8px--s2 图文间距 8px

所有颜色、圆角与间距统一定义于 :root；深浅主题仅覆盖令牌。状态文案通过重叠网格预留空间，不依赖固定文案宽度。Button、copyText 均可直接调用。
